# SKSE-Hotfixes: pinned NG 11 build; no automatic deployment to game folders.
cmake_minimum_required(VERSION 3.21)
project(SummonActorLimitOverhaul VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENABLE_SKYRIM_SE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_AE ON CACHE BOOL "" FORCE)
set(ENABLE_SKYRIM_VR ON CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
if(COMMONLIB_WORKBENCH_CACHE)
  # SKSE-Hotfixes: optional cache from our own source-built clang-cl 21.1.8 /MT library.
  if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "21.1.8" OR NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "MultiThreaded")
    message(FATAL_ERROR "Workbench cache requires clang-cl 21.1.8, Release static CRT.")
  endif()
  file(SHA256 "${COMMONLIB_WORKBENCH_CACHE}/lib/CommonLibSSE.lib" cache_hash)
  if(NOT cache_hash STREQUAL "c1ad860e216ceb7984f55e43bb818c4bb0893f995a6a696ee82853b7e958d6d1")
    message(FATAL_ERROR "Wrong CommonLib cache contents; use a clean source build instead.")
  endif()
  find_package(directxtk CONFIG REQUIRED)
  find_package(CommonLibSSE CONFIG REQUIRED PATHS "${COMMONLIB_WORKBENCH_CACHE}/lib/cmake/CommonLibSSE" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  target_include_directories(CommonLibSSE::CommonLibSSE INTERFACE "$ENV{COMMONLIB_SSE_FOLDER}/extern/openvr/headers")
else()
  add_subdirectory("$ENV{COMMONLIB_SSE_FOLDER}" "${CMAKE_CURRENT_BINARY_DIR}/_deps/clib-build")
  include("$ENV{COMMONLIB_SSE_FOLDER}/cmake/CommonLibSSE.cmake")
endif()
add_commonlibsse_plugin(${PROJECT_NAME}
  SOURCES src/plugin.cpp src/include/PerkEntryPointExtenderAPI.h
  NAME "SummonActorLimitOverhaul" AUTHOR "Asrak" VERSION 1.0)
target_precompile_headers(${PROJECT_NAME} PRIVATE src/PCH.h)
